import requests, time, random

def retry_request(endpoint, api_key, model, max_attempts=6, base_delay=1.0):
    headers = {"Authorization": f"Bearer {api_key}", "Content-Type":"application/json"}
    for attempt in range(1, max_attempts+1):
        try:
            r = requests.post(endpoint, json={"model": model, "input":"your prompt"}, headers=headers, timeout=20)
            if 200 <= r.status_code < 300:
                return r.json()
            status = r.status_code
        except requests.RequestException as e:
            status = str(e)
        if attempt == max_attempts:
            raise Exception(f"Failed after {attempt} attempts. last: {status}")
        jitter = random.uniform(0, 0.5)
        delay = base_delay * (2**(attempt-1)) + jitter
        print(f"Attempt {attempt} -> {status}. Retrying in {delay:.2f}s")
        time.sleep(delay)
